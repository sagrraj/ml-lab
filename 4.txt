import pandas as pd
from sklearn.tree import DecisionTreeClassifier
from sklearn import tree
import matplotlib.pyplot as plt

df = pd.read_csv(r"d:\ML LAB code\DATASET-2026\datasets\Play_Tennis.csv")

df['outlook'] = df['outlook'].map({
    'sunny': 0,
    'overcast': 1,
    'rainy': 2
})

df['temp'] = df['temp'].map({
    'hot': 0,
    'mild': 1,
    'cool': 2
})

df['humidity'] = df['humidity'].map({
    'high': 0,
    'normal': 1
})

df['windy'] = df['windy'].map({
    False: 0,
    True: 1
})

df['play'] = df['play'].map({
    'no': 0,
    'yes': 1
})

X = df[['outlook', 'temp', 'humidity', 'windy']]
y = df['play']

model = DecisionTreeClassifier(criterion='entropy')

model.fit(X, y)

test = pd.DataFrame([[0,1,1,0]],
                    columns=['outlook','temp','humidity','windy'])

prediction = model.predict(test)

print("Prediction:", prediction[0])

plt.figure(figsize=(10,6))

tree.plot_tree(
    model,
    feature_names=['outlook', 'temp', 'humidity', 'windy'],
    class_names=['No', 'Yes'],
)

plt.show()
