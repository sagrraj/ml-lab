import matplotlib.pyplot as plt
import pandas as pd
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler

df = pd.read_csv(r"./iris1.csv")
df.columns = [c.strip() for c in df.columns]
X = df[['Sepal_Length', 'Sepal_Width', 'Petal_Length', 'Petal_Width']].values
print("iris1.csv:", df.shape, "4 features -> 2 PCs")
y = pd.Categorical(df['Class']).codes

X_scaled = StandardScaler().fit_transform(X)
pca = PCA(n_components=2)
X_pca = pca.fit_transform(X_scaled)

plt.figure(figsize=(8,6))
plt.scatter(X_pca[:,0], X_pca[:,1], c=y, cmap='viridis', edgecolors='k')
plt.xlabel('PC1'); plt.ylabel('PC2'); plt.title('PCA of Iris (iris1.csv)')
plt.show()
print("Explained variance:", pca.explained_variance_ratio_)
print("Total retained:", round(pca.explained_variance_ratio_.sum(), 4))
